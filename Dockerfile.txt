FROM python:3.12-slim

WORKDIR /app

# System dependencies
RUN apt-get update && \
    apt-get install -y \
        ffmpeg \
        curl \
        unzip \
        ca-certificates \
        nodejs \
        npm && \
    rm -rf /var/lib/apt/lists/*

# Install Deno
RUN curl -fsSL https://deno.land/install.sh | sh

ENV DENO_INSTALL=/root/.deno
ENV PATH=$DENO_INSTALL/bin:$PATH

# Python dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Download PO token provider
RUN git clone --depth 1 \
    https://github.com/Brainicism/bgutil-ytdlp-pot-provider.git \
    /root/bgutil-ytdlp-pot-provider

# Install provider dependencies
WORKDIR /root/bgutil-ytdlp-pot-provider/server
RUN deno install --allow-scripts=npm:canvas

# Return to app
WORKDIR /app
COPY . .

ENV PYTHONUNBUFFERED=1

# Start PO token provider + Flask
CMD ["sh", "-c", "deno run --allow-env --allow-net --allow-ffi=. --allow-read=. /root/bgutil-ytdlp-pot-provider/server/src/main.ts --host 127.0.0.1 --port 4416 & sleep 3 && gunicorn --bind 0.0.0.0:${PORT:-10000} app:app"]
